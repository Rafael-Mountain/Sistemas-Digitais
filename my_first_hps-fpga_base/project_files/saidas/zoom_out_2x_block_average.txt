// ================================================================================
// Módulo: zoom_out_2x_block_average.v
//
// Descrição: Zoom Out 2x por Média.
// Atualizado: Leitura sequencial com S_WAIT para latência de 2 ciclos.
//             Escrita em 2 ciclos.
// ================================================================================

module zoom_out_2x_block_average (
    input wire clk, reset, start,
    input wire [7:0] rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH = 160, RAM_WIDTH = 80, RAM_HEIGHT = 60;
    
    // Estados expandidos para acomodar latência de leitura
    localparam S_IDLE      = 4'd0;
    localparam S_ADDR_00   = 4'd1; localparam S_WAIT_00 = 4'd2;
    localparam S_ADDR_01   = 4'd3; localparam S_WAIT_01 = 4'd4;
    localparam S_ADDR_10   = 4'd5; localparam S_WAIT_10 = 4'd6;
    localparam S_ADDR_11   = 4'd7; localparam S_WAIT_11 = 4'd8;
    localparam S_CALC      = 4'd9; // Captura o último e calcula
    localparam S_WRITE_1   = 4'd10;
    localparam S_WRITE_2   = 4'd11;
    localparam S_DONE      = 4'd12;

    reg [3:0] state;
    reg [6:0] ram_x; 
    reg [5:0] ram_y;
    reg [7:0] p00, p01, p10, p11;
    reg [9:0] pixel_sum;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE;
            ram_x <= 0; ram_y <= 0; ram_wren <= 0; done <= 0;
            p00<=0; p01<=0; p10<=0; p11<=0; pixel_sum<=0;
        end else begin
            case (state)
                S_IDLE: begin 
                    done <= 0; ram_wren <= 0; ram_x <= 0; ram_y <= 0;
                    if(start) state <= S_ADDR_00;
                end

                // --- Leitura Pixel 00 ---
                S_ADDR_00: begin
                    ram_wren <= 0;
                    state <= S_WAIT_00;
                end
                S_WAIT_00: state <= S_ADDR_01; // Dado 00 estará pronto no próximo ciclo

                // --- Leitura Pixel 01 ---
                S_ADDR_01: begin
                    p00 <= rom_data; // Captura Pixel 00
                    state <= S_WAIT_01;
                end
                S_WAIT_01: state <= S_ADDR_10;

                // --- Leitura Pixel 10 ---
                S_ADDR_10: begin
                    p01 <= rom_data; // Captura Pixel 01
                    state <= S_WAIT_10;
                end
                S_WAIT_10: state <= S_ADDR_11;

                // --- Leitura Pixel 11 ---
                S_ADDR_11: begin
                    p10 <= rom_data; // Captura Pixel 10
                    state <= S_WAIT_11;
                end
                S_WAIT_11: state <= S_CALC;

                // --- Cálculo ---
                S_CALC: begin
                    p11 <= rom_data; // Captura Pixel 11
                    // Soma precisa usar p11 deste ciclo, ou registramos e somamos depois.
                    // Para simplificar, usamos soma combinacional no próximo estado ou aqui:
                    // Vamos registrar a soma para usar no write.
                    pixel_sum <= p00 + p01 + p10 + rom_data; 
                    state <= S_WRITE_1;
                end

                // --- Escrita (2 Ciclos) ---
                S_WRITE_1: begin
                    ram_wren <= 1;
                    ram_data <= pixel_sum >> 2; // Divisão por 4
                    state <= S_WRITE_2;
                end

                S_WRITE_2: begin
                    ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                    ram_data <= pixel_sum >> 2;
                    
                    if (ram_x == RAM_WIDTH - 1) begin
                        ram_x <= 0;
                        if (ram_y == RAM_HEIGHT - 1) state <= S_DONE;
                        else begin ram_y <= ram_y + 1; state <= S_ADDR_00; end
                    end else begin
                        ram_x <= ram_x + 1;
                        state <= S_ADDR_00;
                    end
                end

                S_DONE: begin
                    ram_wren <= 0; done <= 1;
                    if(!start) state <= S_IDLE;
                end
            endcase
        end
    end

    // Lógica Combinacional de Endereços
    always @(*) begin
        // Endereço de Destino
        ram_addr = (ram_y * RAM_WIDTH) + ram_x;
        
        // Endereço de Origem (Mapeado por Estado)
        rom_addr = 17'd0;
        case(state)
            S_ADDR_00: rom_addr = ((ram_y*2)+0)*ROM_WIDTH + ((ram_x*2)+0);
            S_ADDR_01: rom_addr = ((ram_y*2)+0)*ROM_WIDTH + ((ram_x*2)+1);
            S_ADDR_10: rom_addr = ((ram_y*2)+1)*ROM_WIDTH + ((ram_x*2)+0);
            S_ADDR_11: rom_addr = ((ram_y*2)+1)*ROM_WIDTH + ((ram_x*2)+1);
            default:   rom_addr = 17'd0;
        endcase
    end
endmodule
