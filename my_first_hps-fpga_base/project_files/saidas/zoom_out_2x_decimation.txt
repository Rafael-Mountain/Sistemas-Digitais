// Módulo: zoom_out_2x_decimation.v
//
// Descrição: Zoom Out 2x por Dizimação.
// Atualizado: Substituição de Pipeline por FSM com S_WAIT e Escrita de 2 Ciclos.
// ================================================================================

module zoom_out_2x_decimation (
    input wire clk, reset, start,
    input  wire [7:0]  rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH = 160, RAM_WIDTH = 80, ROM_HEIGHT = 120;
    localparam S_IDLE=0, S_READ=1, S_WAIT=2, S_WRITE_1=3, S_WRITE_2=4, S_DONE=5;
    reg [2:0] state;
    
    reg [7:0] ram_x; 
    reg [6:0] ram_y;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE;
            ram_x <= 0; ram_y <= 0; ram_wren <= 0; done <= 0;
        end else begin
            case(state)
                S_IDLE: begin
                    done <= 0; ram_wren <= 0; ram_x <= 0; ram_y <= 0;
                    if(start) state <= S_READ;
                end

                S_READ: begin
                    ram_wren <= 0;
                    // Endereço de origem colocado na saída combinacional
                    state <= S_WAIT;
                end

                S_WAIT: begin
                    // Espera latência da RAM
                    state <= S_WRITE_1;
                end

                S_WRITE_1: begin
                    ram_wren <= 1;
                    ram_data <= rom_data; // Dado válido
                    state <= S_WRITE_2;
                end

                S_WRITE_2: begin
                    ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                    ram_data <= rom_data;
                    
                    // Atualiza contadores de destino
                    if (ram_x == RAM_WIDTH - 1) begin
                        ram_x <= 0;
                        // Altura destino = 60 (ROM_HEIGHT/2)
                        if (ram_y == (ROM_HEIGHT/2) - 1) 
                            state <= S_DONE;
                        else begin 
                            ram_y <= ram_y + 1; 
                            state <= S_READ; 
                        end
                    end else begin
                        ram_x <= ram_x + 1;
                        state <= S_READ;
                    end
                end

                S_DONE: begin
                    ram_wren <= 0; done <= 1;
                    if(!start) state <= S_IDLE;
                end
            endcase
        end
    end

    always @(*) begin
        // Pega pixels pares: (x*2, y*2)
        rom_addr = ((ram_y * 2) * ROM_WIDTH) + (ram_x * 2);
        ram_addr = (ram_y * RAM_WIDTH) + ram_x;
    end
endmodule
