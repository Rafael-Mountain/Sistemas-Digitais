// ================================================================================
// Módulo: zoom_in_4x_replication.v
//
// Descrição: Zoom In 4x por Replicação.
// Atualizado: Leitura com latência (S_WAIT) e Escrita Estável de 2 ciclos.
// ================================================================================

module zoom_in_4x_replication (
    input wire clk, reset, start,
    input  wire [7:0]  rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH = 160, RAM_WIDTH = 640, ROM_HEIGHT = 120;
    localparam S_IDLE=0, S_READ=1, S_WAIT=2, S_WRITE_1=3, S_WRITE_2=4, S_DONE=5;
    reg [2:0] state;
    
    reg [7:0] rom_x;        // X Origem
    reg [6:0] rom_y;        // Y Origem
    reg [3:0] write_state;  // Contador de sub-pixel (0 a 15 para bloco 4x4)
    reg [7:0] pixel_buffer;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE;
            rom_x <= 0; rom_y <= 0; write_state <= 0; 
            pixel_buffer <= 0; ram_wren <= 0; done <= 0;
        end else begin
            case(state)
                S_IDLE: begin
                    ram_wren <= 0; done <= 0; 
                    rom_x <= 0; rom_y <= 0; write_state <= 0;
                    if(start) state <= S_READ;
                end

                S_READ: begin
                    ram_wren <= 0;
                    state <= S_WAIT;
                end

                S_WAIT: begin
                    // Espera RAM
                    state <= S_WRITE_1;
                end

                S_WRITE_1: begin
                    ram_wren <= 1;
                    // Captura dado no primeiro sub-pixel
                    if (write_state == 0) pixel_buffer <= rom_data;
                    
                    ram_data <= (write_state == 0) ? rom_data : pixel_buffer;
                    state <= S_WRITE_2;
                end

                S_WRITE_2: begin
                    ram_wren <= 1;
                    ram_data <= pixel_buffer;
                    
                    // Lógica de Loop do Bloco 4x4
                    if (write_state == 15) begin
                        // Fim do bloco 4x4
                        ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                        write_state <= 0;
                        
                        if (rom_x == ROM_WIDTH - 1) begin
                            rom_x <= 0;
                            if (rom_y == ROM_HEIGHT - 1) 
                                state <= S_DONE;
                            else begin 
                                rom_y <= rom_y + 1; 
                                state <= S_READ; 
                            end
                        end else begin
                            rom_x <= rom_x + 1;
                            state <= S_READ;
                        end
                    end else begin
                        // Próximo sub-pixel
                        write_state <= write_state + 1;
                        state <= S_WRITE_1;
                    end
                end

                S_DONE: begin
                    ram_wren <= 0; done <= 1;
                    if(!start) state <= S_IDLE;
                end
            endcase
        end
    end

    always @(*) begin
        reg [1:0] sy, sx;
        // Endereço Origem
        rom_addr = (rom_y * ROM_WIDTH) + rom_x;
        
        // Endereço Destino (Mapeamento 4x4)
        sy = write_state / 4; 
        sx = write_state % 4;
        ram_addr = ((rom_y*4 + sy) * RAM_WIDTH) + (rom_x*4 + sx);
    end
endmodule
