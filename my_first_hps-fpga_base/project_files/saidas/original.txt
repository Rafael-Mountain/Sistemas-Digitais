// ================================================================================
// Módulo: original.v
//
// Descrição: Copia imagem e limpa o resto da memória.
// Atualizado: Leitura com latência de 2 ciclos e Escrita em 2 ciclos.
// ================================================================================

module original (
    input wire clk, reset, start,
    input  wire [7:0]  rom_data,
    output reg  [16:0] rom_addr,
    output reg  [18:0] ram_addr,
    output reg  [7:0]  ram_data,
    output reg         ram_wren,
    output reg         done
);
    localparam IMAGE_SIZE = 19200;
    localparam RAM_SIZE_TO_CLEAR = 307200;

    localparam S_IDLE=0, S_READ=1, S_WAIT=2, S_WRITE_1=3, S_WRITE_2=4, S_DONE=5;
    reg [2:0] state;
    reg [18:0] addr_counter;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE;
            addr_counter <= 0;
            ram_wren <= 0; done <= 0;
        end else begin
            case(state)
                S_IDLE: begin
                    ram_wren <= 0; done <= 0; addr_counter <= 0;
                    if(start) state <= S_READ;
                end
                S_READ: begin
                    ram_wren <= 0;
                    // Endereço é colocado (lógica combinacional abaixo)
                    state <= S_WAIT;
                end
                S_WAIT: begin
                    // Espera latência da RAM (Ciclo extra)
                    ram_wren <= 0;
                    state <= S_WRITE_1;
                end
                S_WRITE_1: begin
                    ram_wren <= 1;
                    // Se dentro da imagem, usa rom_data (agora válido). Senão, preto.
                    ram_data <= (addr_counter < IMAGE_SIZE) ? rom_data : 8'd0;
                    state <= S_WRITE_2;
                end
                S_WRITE_2: begin
                    ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                    ram_data <= (addr_counter < IMAGE_SIZE) ? rom_data : 8'd0;
                    
                    if(addr_counter == RAM_SIZE_TO_CLEAR - 1) begin
                        state <= S_DONE;
                    end else begin
                        addr_counter <= addr_counter + 1;
                        state <= S_READ;
                    end
                end
                S_DONE: begin
                    ram_wren <= 0; done <= 1;
                    if(!start) state <= S_IDLE;
                end
            endcase
        end
    end

    // Lógica Combinacional de Endereço
    always @(*) begin
        rom_addr = addr_counter[16:0];
        ram_addr = addr_counter;
    end
endmodule
