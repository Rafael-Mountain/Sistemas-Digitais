// original.v
module original (
    input wire clk,
    input wire reset,
    input wire start,
    output reg  [16:0] rom_addr,
    input  wire [7:0]  rom_data,
    output reg  [18:0] ram_addr,
    output reg  [7:0]  ram_data,
    output reg         ram_wren,
    output reg         done
);

    localparam IMAGE_SIZE = 19200; // 160 * 120
    localparam RAM_SIZE_TO_CLEAR = 307200; // 640 * 480

    reg [18:0] addr_counter;
    reg processing;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            addr_counter <= 0;
            ram_wren <= 1'b0;
            done <= 1'b0;
            processing <= 1'b0;
        end else begin
            if (start && !processing) begin
                processing <= 1'b1;
                addr_counter <= 0;
                done <= 1'b0;
            end else if (processing) begin
                ram_wren <= 1'b1;
                
                if (addr_counter == RAM_SIZE_TO_CLEAR - 1) begin
                    processing <= 1'b0;
                    done <= 1'b1;
                    ram_wren <= 1'b0;
                end else begin
                    addr_counter <= addr_counter + 1;
                end
            end else begin
                 ram_wren <= 1'b0;
                 done <= 1'b0;
            end
        end
    end

    always @(*) begin
        rom_addr = addr_counter[16:0];
        ram_addr = addr_counter;

        if (addr_counter < IMAGE_SIZE) begin
            ram_data = rom_data;
        end else begin
            ram_data = 8'd0; // Limpa o restante da RAM
        end
    end

endmodule
