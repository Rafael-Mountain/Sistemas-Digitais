// ================================================================================
// Módulo: zoom_out_4x_block_average.v
//
// Descrição: Zoom Out 4x por Média.
// Atualizado: Loop de leitura de 16 pixels com suporte a latência de RAM.
//             Escrita em 2 ciclos.
// ================================================================================

module zoom_out_4x_block_average (
    input wire clk, reset, start,
    input wire [7:0] rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH = 160, RAM_WIDTH = 40, RAM_HEIGHT = 30;
    
    // Estados da FSM
    localparam S_IDLE      = 3'd0;
    localparam S_SET_ADDR  = 3'd1; // Coloca endereço na ROM
    localparam S_WAIT_RAM  = 3'd2; // Espera latência
    localparam S_STORE     = 3'd3; // Captura dado no buffer
    localparam S_CALC      = 3'd4; // Soma tudo
    localparam S_WRITE_1   = 3'd5; // Escrita ciclo 1
    localparam S_WRITE_2   = 3'd6; // Escrita ciclo 2
    localparam S_DONE      = 3'd7;

    reg [2:0] state;
    
    reg [5:0] ram_x; 
    reg [4:0] ram_y;
    
    reg [4:0] read_counter; // Conta de 0 a 15 (pixels do bloco 4x4)
    reg [7:0] p_buf [0:15]; // Buffer
    reg [11:0] pixel_sum;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE;
            ram_x <= 0; ram_y <= 0; ram_wren <= 0; done <= 0;
            read_counter <= 0; pixel_sum <= 0;
        end else begin
            case (state)
                S_IDLE: begin 
                    done <= 0; ram_wren <= 0; ram_x <= 0; ram_y <= 0; read_counter <= 0;
                    if(start) state <= S_SET_ADDR;
                end

                // --- Loop de Leitura (16x) ---
                S_SET_ADDR: begin
                    ram_wren <= 0;
                    // Endereço é calculado combinacionalmente baseado em read_counter
                    state <= S_WAIT_RAM;
                end

                S_WAIT_RAM: begin
                    // Espera latência da RAM CLOCK0
                    state <= S_STORE;
                end

                S_STORE: begin
                    // Captura o pixel
                    p_buf[read_counter] <= rom_data;
                    
                    if (read_counter == 16'd15) begin
                        read_counter <= 0; // Reseta para o próximo bloco
                        state <= S_CALC;
                    end else begin
                        read_counter <= read_counter + 1;
                        state <= S_SET_ADDR; // Lê próximo pixel
                    end
                end

                // --- Cálculo ---
                S_CALC: begin
                    pixel_sum <= p_buf[0] + p_buf[1] + p_buf[2] + p_buf[3] +
                                 p_buf[4] + p_buf[5] + p_buf[6] + p_buf[7] +
                                 p_buf[8] + p_buf[9] + p_buf[10] + p_buf[11] +
                                 p_buf[12] + p_buf[13] + p_buf[14] + p_buf[15];
                    state <= S_WRITE_1;
                end

                // --- Escrita (2 Ciclos) ---
                S_WRITE_1: begin
                    ram_wren <= 1;
                    ram_data <= pixel_sum >> 4; // Divisão por 16
                    state <= S_WRITE_2;
                end

                S_WRITE_2: begin
                    ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                    ram_data <= pixel_sum >> 4;

                    if (ram_x == RAM_WIDTH - 1) begin
                        ram_x <= 0;
                        if (ram_y == RAM_HEIGHT - 1) state <= S_DONE;
                        else begin ram_y <= ram_y + 1; state <= S_SET_ADDR; end
                    end else begin
                        ram_x <= ram_x + 1;
                        state <= S_SET_ADDR;
                    end
                end

                S_DONE: begin
                    ram_wren <= 0; done <= 1;
                    if(!start) state <= S_IDLE;
                end
            endcase
        end
    end

    // Lógica Combinacional de Endereços
    always @(*) begin
        ram_addr = (ram_y * RAM_WIDTH) + ram_x;
        
        // Endereço de Origem baseado no contador do loop (read_counter)
        begin
            reg [1:0] sy, sx;
            sy = read_counter / 4;
            sx = read_counter % 4;
            rom_addr = ((ram_y * 4) + sy) * ROM_WIDTH + ((ram_x * 4) + sx);
        end
    end
endmodule
