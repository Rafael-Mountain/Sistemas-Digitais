//
// Descrição: Zoom In 2x NN.
// Atualizado: Leitura com latência de 2 ciclos, Escrita em 2 ciclos.
// ================================================================================

module zoom_in_2x_nearest_neighbor (
    input wire clk, reset, start,
    input  wire [7:0]  rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH=160, RAM_WIDTH=320, RAM_HEIGHT=240;
    
    localparam S_IDLE=0, S_READ_ROM=1, S_WAIT=2, S_WRITE_1=3, S_WRITE_2=4, S_DONE=5;
    reg [2:0] state, next_state;

    reg [8:0] ram_x; 
    reg [7:0] ram_y; 
    reg [7:0] pixel_buffer;

    always @(posedge clk or posedge reset) begin
        if (reset) state <= S_IDLE;
        else state <= next_state;
    end

    always @(posedge clk) begin
        if (reset) begin
            ram_x <= 0; ram_y <= 0; ram_wren <= 0; done <= 0; pixel_buffer <= 0;
        end else begin
            case (state)
                S_IDLE: begin 
                    done <= 0; ram_wren <= 0; ram_x <= 0; ram_y <= 0; 
                end
                S_READ_ROM: begin
                    ram_wren <= 0; 
                end
                S_WAIT: begin
                    ram_wren <= 0; // Espera dado
                end
                S_WRITE_1: begin
                    ram_wren <= 1;
                    pixel_buffer <= rom_data; // Captura dado válido
                    ram_data <= rom_data;
                end
                S_WRITE_2: begin
                    ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                    ram_data <= pixel_buffer; // Mantém dado estável
                    
                    if (ram_x == RAM_WIDTH - 1) begin 
                        ram_x <= 0; 
                        ram_y <= (ram_y == RAM_HEIGHT - 1) ? 0 : ram_y + 1;
                    end else begin
                        ram_x <= ram_x + 1; 
                    end
                end
                S_DONE: begin 
                    ram_wren <= 0; done <= 1; 
                end
            endcase
        end
    end

    always @(*) begin
        next_state = state;
        rom_addr = ((ram_y >> 1) * ROM_WIDTH) + (ram_x >> 1);
        ram_addr = (ram_y * RAM_WIDTH) + ram_x;

        case (state)
            S_IDLE:     if (start) next_state = S_READ_ROM;
            S_READ_ROM: next_state = S_WAIT;
            S_WAIT:     next_state = S_WRITE_1;
            S_WRITE_1:  next_state = S_WRITE_2;
            S_WRITE_2:  if (ram_x == RAM_WIDTH - 1 && ram_y == RAM_HEIGHT - 1) 
                            next_state = S_DONE; 
                        else 
                            next_state = S_READ_ROM; 
            S_DONE:     if (!start) next_state = S_IDLE;
        endcase
    end
endmodule
