// ================================================================================
// Módulo: zoom_in_2x_replication.v
//
// Descrição: Zoom In 2x por Replicação.
// Atualizado: Leitura com latência (S_WAIT) e Escrita Estável de 2 ciclos por sub-pixel.
// ================================================================================

module zoom_in_2x_replication (
    input wire clk, reset, start,
    input  wire [7:0]  rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH = 160, RAM_WIDTH = 320, ROM_HEIGHT = 120;
    
    // Estados: READ -> WAIT -> WRITE_1 -> WRITE_2 (Loop para os 4 pixels) -> DONE
    localparam S_IDLE=0, S_READ=1, S_WAIT=2, S_WRITE_1=3, S_WRITE_2=4, S_DONE=5;
    reg [2:0] state;
    
    reg [7:0] rom_x;        // Coordenada X origem
    reg [6:0] rom_y;        // Coordenada Y origem
    reg [1:0] write_state;  // Contador de sub-pixel (0 a 3)
    reg [7:0] pixel_buffer; // Armazena o pixel lido

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE;
            rom_x <= 0; rom_y <= 0; write_state <= 0; 
            pixel_buffer <= 0; ram_wren <= 0; done <= 0;
        end else begin
            case(state)
                S_IDLE: begin
                    ram_wren <= 0; done <= 0; 
                    rom_x <= 0; rom_y <= 0; write_state <= 0;
                    if(start) state <= S_READ;
                end

                S_READ: begin
                    ram_wren <= 0;
                    // Endereço de leitura já está no barramento.
                    state <= S_WAIT;
                end

                S_WAIT: begin
                    // Espera o dado propagar (latência da RAM CLOCK0)
                    state <= S_WRITE_1;
                end

                S_WRITE_1: begin
                    // Ciclo 1 de escrita do sub-pixel atual
                    ram_wren <= 1;
                    // Se viemos de S_WAIT, capturamos o dado agora.
                    // Se viemos de S_WRITE_2 (loop), pixel_buffer já tem o dado.
                    if (write_state == 0) pixel_buffer <= rom_data;
                    
                    // Nota: ram_data recebe pixel_buffer (ou rom_data no 1º sub-pixel)
                    ram_data <= (write_state == 0) ? rom_data : pixel_buffer;
                    
                    state <= S_WRITE_2;
                end

                S_WRITE_2: begin
                    // Ciclo 2 de escrita do sub-pixel atual
                    ram_wren <= 1;
                    ram_data <= pixel_buffer;
                    
                    // Lógica de Loop do Bloco 2x2
                    if (write_state == 3) begin
                        // Terminou o bloco 2x2
                        ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                        write_state <= 0;
                        
                        // Avança na imagem de origem
                        if (rom_x == ROM_WIDTH - 1) begin
                            rom_x <= 0;
                            if (rom_y == ROM_HEIGHT - 1) 
                                state <= S_DONE;
                            else begin 
                                rom_y <= rom_y + 1; 
                                state <= S_READ; 
                            end
                        end else begin
                            rom_x <= rom_x + 1;
                            state <= S_READ;
                        end
                    end else begin
                        // Próximo sub-pixel do mesmo bloco
                        write_state <= write_state + 1;
                        state <= S_WRITE_1; // Volta para escrever o próximo
                    end
                end

                S_DONE: begin
                    ram_wren <= 0; done <= 1;
                    if(!start) state <= S_IDLE;
                end
            endcase
        end
    end

    // Lógica Combinacional de Endereços
    always @(*) begin
        // Endereço de Origem
        rom_addr = (rom_y * ROM_WIDTH) + rom_x;
        
        // Endereço de Destino (Baseado no sub-pixel atual)
        case(write_state)
            2'b00: ram_addr = ((rom_y * 2) * RAM_WIDTH) + (rom_x * 2);      // (0,0)
            2'b01: ram_addr = ((rom_y * 2) * RAM_WIDTH) + (rom_x * 2 + 1);  // (1,0)
            2'b10: ram_addr = ((rom_y * 2 + 1) * RAM_WIDTH) + (rom_x * 2);  // (0,1)
            default: ram_addr = ((rom_y * 2 + 1) * RAM_WIDTH) + (rom_x * 2 + 1); // (1,1)
        endcase
    end
endmodule
