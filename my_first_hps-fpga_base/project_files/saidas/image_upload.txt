//
// Correção v6: Ajuste no mapeamento de bits para evitar conflito com Opcode.
//              Endereço movido para bits [22:8].
// ================================================================================

module image_upload (
    input wire clk,
    input wire reset,
    input wire start,
    input wire [31:0] instruction,

    output reg [14:0] ram_image_addr,
    output reg [7:0]  ram_image_data,
    output reg        ram_image_wren,

    output reg [18:0] ram_op_addr,
    output reg [7:0]  ram_op_data,
    output reg        ram_op_wren,

    output reg        done
);

    // --- CORREÇÃO DE MAPEAMENTO DE BITS ---
    // Estrutura da Instrução:
    // [31:28] Opcode
    // [27:23] Unused
    // [22:08] Address (15 bits) <--- MUDANÇA AQUI (Era [28:14])
    // [07:00] Data (8 bits)
    
    wire [14:0] pixel_addr = instruction[22:8];
    wire [7:0]  pixel_data = instruction[7:0];

    // FSM de Escrita em 2 Ciclos
    localparam S_IDLE    = 2'd0;
    localparam S_WRITE_1 = 2'd1;
    localparam S_WRITE_2 = 2'd2;
    localparam S_DONE    = 2'd3;

    reg [1:0] state, next_state;

    always @(posedge clk or posedge reset) begin
        if (reset) state <= S_IDLE;
        else state <= next_state;
    end

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            ram_image_wren <= 0; ram_image_addr <= 0; ram_image_data <= 0;
            ram_op_wren <= 0; ram_op_addr <= 0; ram_op_data <= 0;
            done <= 0;
        end else begin
            case (state)
                S_IDLE: begin
                    ram_image_wren <= 0; ram_op_wren <= 0; done <= 0;
                end

                S_WRITE_1: begin
                    // Configura endereços e dados
                    ram_image_addr <= pixel_addr;
                    ram_image_data <= pixel_data;
                    ram_image_wren <= 1;

                    ram_op_addr    <= {4'b0, pixel_addr}; 
                    ram_op_data    <= pixel_data;
                    ram_op_wren    <= 1;
                    
                    done <= 0;
                end

                S_WRITE_2: begin
                    // Mantém escrita por mais um ciclo (Estabilidade)
                    ram_image_wren <= 1;
                    ram_op_wren    <= 1;
                    done <= 0;
                end

                S_DONE: begin
                    ram_image_wren <= 0; ram_op_wren <= 0;
                    done <= 1;
                end
            endcase
        end
    end

    always @(*) begin
        case (state)
            S_IDLE:    if (start) next_state = S_WRITE_1; else next_state = S_IDLE;
            S_WRITE_1: next_state = S_WRITE_2;
            S_WRITE_2: next_state = S_DONE;
            S_DONE:    if (!start) next_state = S_IDLE; else next_state = S_DONE;
            default:   next_state = S_IDLE;
        endcase
    end
endmodule
