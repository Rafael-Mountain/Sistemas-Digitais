// ================================================================================
// main.v - MÓDULO TOPLEVEL FINAL E COMPLETO
//
// Descrição:
// Versão final que integra o sistema de processamento de imagem e o display de
// depuração. Utiliza um único PIO para instruções e debug, com sincronização
// seletiva aplicada apenas ao barramento crítico para garantir robustez.
// ================================================================================

module main (
    // Entradas Globais
    input wire clock_50,
    input wire reset_button,

    // Entradas Assíncronas do HPS
    input wire [31:0] hps_pio_bus_in,
    input wire        debug_display_select,

    // Saídas para o VGA
    output wire hsync,
    output wire vsync,
    output wire [7:0] red, green, blue,
    output wire sync, clk_out, blank,
	 
	 output wire zoom_in_algo_select_wire,
	 output wire zoom_out_algo_select_wire,
	 
    // Saídas para os Displays de 7 Segmentos (Debug)
    output wire [6:0] seg0_output, seg1_output, seg2_output,
    output wire [6:0] seg3_output, seg4_output, seg5_output
);

    //============================================================================
    // FIOS INTERNOS (WIRES)
    //============================================================================

    // Sinais de Clock e Reset
    wire clock_25;
    wire reset_sync;
    
    // Fio para o barramento de instrução JÁ SINCRONIZADO
    wire [31:0] hps_pio_bus_sync;
    
    // Fios do Sistema de Imagem
    wire program_state, processing_done, instruction_strobe;
    wire [2:0] operation, display_mode;
    wire [18:0] proc_ram_address, video_ram_address, final_ram_address;
    wire [16:0] proc_rom_address;
    wire [7:0] proc_ram_data_in, proc_rom_q_out, ram_q_out;
    wire proc_ram_wren, final_ram_wren;

    //============================================================================
    // INSTÂNCIA DOS MÓDULOS
    //============================================================================

    //----------------------------------------------------------------------------
    // Seção 1: Módulos Base e Sincronizador
    //----------------------------------------------------------------------------
    clock_divider clock_divider_inst ( .clock_in(clock_50), .clock_out(clock_25) );
    sync_reset_button sync_reset_button_inst ( .clock(clock_25), .reset_button_in(reset_button), .reset_sync_out(reset_sync) );

    // Instância do sincronizador APENAS para o barramento crítico de 32 bits.
    synchronizer #( .WIDTH(32) ) pio_bus_synchronizer (
        .clk(clock_25),
        .reset(reset_sync),
        .data_in(hps_pio_bus_in),
        .data_out(hps_pio_bus_sync)
    );

    //----------------------------------------------------------------------------
    // Seção 2: Núcleo de Processamento de Imagem
    //----------------------------------------------------------------------------

    instruction_strobe_detector strobe_inst (
        .clock(clock_25),
        .reset(reset_sync),
        .instruction_bus(hps_pio_bus_sync),
        .strobe_out(instruction_strobe)
    );

    control_unit control_unit_inst (
        .clock(clock_25),
        .reset(reset_sync),
        .instruction_in(hps_pio_bus_sync),
        .instruction_strobe(instruction_strobe),
        .processing_done(processing_done),
        .program_state(program_state), 
        .operation(operation),
        .display_mode(display_mode),
        .zoom_in_algo_select(zoom_in_algo_select_wire),
        .zoom_out_algo_select(zoom_out_algo_select_wire)
    );
    
    processing_unit processing_unit_inst (
        .clock(clock_25),
        .reset(reset_sync),
        .zoom_in_algo_select(zoom_in_algo_select_wire),
        .zoom_out_algo_select(zoom_out_algo_select_wire),
        .program_state(program_state),
        .operation(operation),
        .rom_q_out_in(proc_rom_q_out),
        .rom_addr_out(proc_rom_address),
        .ram_addr_out(proc_ram_address),
        .ram_data_out(proc_ram_data_in),
        .ram_wren_out(proc_ram_wren),
        .processing_done(processing_done)
    );
    
    assign final_ram_address = (program_state == 1'b1) ? proc_ram_address : video_ram_address;
    assign final_ram_wren    = (program_state == 1'b1) ? proc_ram_wren    : 1'b0;

    memory_module mem_inst (
        .clock(clock_25), 
        .reset(reset_sync),
        .ram_address(final_ram_address),
        .ram_data_in(proc_ram_data_in),
        .wren_in(final_ram_wren),
        .q_out_ram(ram_q_out),
        .rom_address_ext(proc_rom_address),
        .q_out_rom(proc_rom_q_out)
    );

    video_controller video_inst (
        .clk(clock_25),
        .reset(reset_sync),
        .display_mode(display_mode),
        .ram_data_in(ram_q_out),           
        .processing_active(program_state), 
        .ram_addr_out(video_ram_address),   
        .hsync(hsync), .vsync(vsync), .red(red), .green(green), 
        .blue(blue), .sync(sync), .clk_out(clk_out), .blank(blank)
    );

    //----------------------------------------------------------------------------
    // Seção 3: Controlador do Display de 7 Segmentos (Debug)
    //----------------------------------------------------------------------------

    top_module seven_segment_controller_inst (
        .hps_pio_data   (hps_pio_bus_sync),
        .display_select (debug_display_select),
        .seg0_output    (seg0_output),
        .seg1_output    (seg1_output),
        .seg2_output    (seg2_output),
        .seg3_output    (seg3_output),
        .seg4_output    (seg4_output),
        .seg5_output    (seg5_output)
    );

endmodule
