// Descrição: Zoom In 4x NN.
// Atualizado: Leitura com latência de 2 ciclos, Escrita em 2 ciclos.
// ================================================================================

module zoom_in_4x_nearest_neighbor (
    input wire clk, reset, start,
    input  wire [7:0]  rom_data,
    output reg [16:0] rom_addr,
    output reg [18:0] ram_addr,
    output reg [7:0]  ram_data,
    output reg        ram_wren,
    output reg        done
);
    localparam ROM_WIDTH=160, RAM_WIDTH=640, RAM_HEIGHT=480;
    
    localparam S_IDLE=0, S_READ_ROM=1, S_WAIT=2, S_WRITE_1=3, S_WRITE_2=4, S_DONE=5;
    reg [2:0] state;

    reg [9:0] ram_x; 
    reg [8:0] ram_y;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= S_IDLE; ram_x <= 0; ram_y <= 0; ram_wren <= 0; done <= 0;
        end else begin
            case (state)
                S_IDLE: begin 
                    done <= 0; ram_wren <= 0; ram_x <= 0; ram_y <= 0; 
                end
                S_READ_ROM: begin 
                    ram_wren <= 0; // Endereço calculado combinacionalmente
                end
                S_WAIT: begin
                    ram_wren <= 0; // Espera RAM
                end
                S_WRITE_1: begin
                    ram_wren <= 1;
                    ram_data <= rom_data; // Dado válido aqui
                end
                S_WRITE_2: begin
                    ram_wren <= 0; // Alterado para 0 para evitar escrita fantasma no próximo ciclo
                    ram_data <= rom_data;
                    
                    if (ram_x == RAM_WIDTH - 1) begin
                        ram_x <= 0; 
                        ram_y <= (ram_y == RAM_HEIGHT - 1) ? 0 : ram_y + 1;
                    end else begin
                        ram_x <= ram_x + 1;
                    end
                end
                S_DONE: begin 
                    ram_wren <= 0; done <= 1; 
                end
                default: ram_wren <= 0;
            endcase
            
            // Lógica de Próximo Estado Integrada
            case (state)
                S_IDLE:     if (start) state <= S_READ_ROM;
                S_READ_ROM: state <= S_WAIT;
                S_WAIT:     state <= S_WRITE_1;
                S_WRITE_1:  state <= S_WRITE_2;
                S_WRITE_2:  if (ram_x == RAM_WIDTH - 1 && ram_y == RAM_HEIGHT - 1) 
                                state <= S_DONE; 
                            else 
                                state <= S_READ_ROM;
                S_DONE:     if (!start) state <= S_IDLE;
                default:    state <= S_IDLE;
            endcase
        end
    end

    always @(*) begin
        rom_addr = ((ram_y >> 2) * ROM_WIDTH) + (ram_x >> 2);
        ram_addr = (ram_y * RAM_WIDTH) + ram_x;
    end
endmodule
